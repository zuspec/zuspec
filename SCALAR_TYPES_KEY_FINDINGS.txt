def test_type_int(parser):
    """Test int type — verify DataTypeInt and location"""
    code = """
struct test_s {
    rand int value;
};
"""
    root = parse_pss(code, parser=parser)
    sym = get_symbol(root, "test_s")
    field = sym.getChild(sym.symtabAt("value"))
    assert isinstance(field.getType(), ast.DataTypeInt)
    loc = get_location(sym.getTarget())
    assert loc is not None
    assert loc[0] == 2

AST→IR TEST PATTERN (test_ast_to_ir.py):

class TestAstToIr(unittest.TestCase):
    def setUp(self):
        self.parser = Parser()
        self.translator = AstToIrTranslator(debug=False)
    
    def parse_and_translate(self, pss_code: str) -> AstToIrContext:
        self.parser.parses([(filename, pss_code)])
        ast_root = self.parser.link()
        ctx = self.translator.translate(ast_root)
        return ctx
    
    def test_builtin_types(self):
        ctx = self.parse_and_translate("component C {}")
        assert "bool" in ctx.type_map
        bool_type = ctx.type_map["bool"]
        assert isinstance(bool_type, ir.DataTypeInt)
        assert bool_type.bits == 1
        assert bool_type.signed == False
